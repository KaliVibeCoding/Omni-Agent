## **🤖 PART 9: AUTOMATED SCRAPING SCRIPT**


### **Universal Academic Paper Scraper**


```python
# universal_academic_scraper.py


import requests
import xml.etree.ElementTree as ET
from typing import List, Dict
import time
import json
from concurrent.futures import ThreadPoolExecutor
import os


class UniversalAcademicScraper: """
   Scrape all major academic databases for papers
   """
  
   def __init__(self):
       self.session = requests.Session()
       self.results = {
   "arxiv": [],
   "semantic_scholar": [],
   "pubmed": [],
   "openalex": [],
   "papers_with_code": []
}
  
   def scrape_arxiv(self, query: str, max_results: int = 10000): """
       Scrape arXiv for papers
       """
      
       print(f"🔍 Scraping arXiv for: {query}")
      
       base_url = "http://export.arxiv.org/api/query"
       params = {
   "search_query": f"all:{query}",
   "start": 0,
   "max_results": max_results,
   "sortBy": "submittedDate",
   "sortOrder": "descending"
}
      
       response = self.session.get(base_url, params=params)
      
       # Parse XML
       root = ET.fromstring(response.content)
      
       papers = []
       for entry in root.findall('{http: //www.w3.org/2005/Atom}entry'):
           paper = {
       "id": entry.find('{http: //www.w3.org/2005/Atom}id').text,
           "title": entry.find('{http: //www.w3.org/2005/Atom}title').text,
               "summary": entry.find('{http: //www.w3.org/2005/Atom}summary').text,
                   "published": entry.find('{http: //www.w3.org/2005/Atom}published').text,
                       "authors": [author.find('{http: //www.w3.org/2005/Atom}name').text
                          for author in entry.findall('{http: //www.w3.org/2005/Atom}author')],
                                   "pdf_url": None
                               }
          
           # Get PDF URL
           for link in entry.findall('{http: //www.w3.org/2005/Atom}link'):
               if link.get('title') == 'pdf':
                   paper['pdf_url'
                                   ] = link.get('href')
          
           papers.append(paper)
      
       self.results[
                                       "arxiv"
                                   ] = papers
       print(f"✅ Found {len(papers)} papers on arXiv")
      
       return papers
  
   def scrape_semantic_scholar(self, query: str, limit: int = 1000): """
       Scrape Semantic Scholar
       """
      
       print(f"🔍 Scraping Semantic Scholar for: {query}")
      
       url = "https://api.semanticscholar.org/graph/v1/paper/search"
       params = {
                                       "query": query,
                                       "limit": 100,
                                       "fields": "title,abstract,authors,year,citationCount,influentialCitationCount,url,openAccessPdf"
                                   }
      
       papers = []
       offset = 0
      
       while len(papers) < limit:
           params[
                                       "offset"
                                   ] = offset
          
           response = self.session.get(url, params=params)
           data = response.json()
          
           if "data" not in data or not data[
                                       "data"
                                   ]:
               break
          
           papers.extend(data[
                                       "data"
                                   ])
           offset += 100
          
           time.sleep(1)  # Rate limiting
      
       self.results[
                                       "semantic_scholar"
                                   ] = papers
       print(f"✅ Found {len(papers)} papers on Semantic Scholar")
      
       return papers
  
   def scrape_openalex(self, query: str, per_page: int = 200): """
       Scrape OpenAlex
       """
      
       print(f"🔍 Scraping OpenAlex for: {query}")
      
       url = "https://api.openalex.org/works"
       params = {
                                       "search": query,
                                       "per_page": per_page,
                                       "page": 1
                                   }
      
       papers = []
      
       while True:
           response = self.session.get(url, params=params)
           data = response.json()
          
           if not data.get("results"):
               break
          
           papers.extend(data[
                                       "results"
                                   ])
          
           if not data.get("meta",
                                   {}).get("next_cursor"):
               break
          
           params[
                                       "cursor"
                                   ] = data[
                                       "meta"
                                   ][
                                       "next_cursor"
                                   ]
           time.sleep(0.1)
      
       self.results[
                                       "openalex"
                                   ] = papers
       print(f"✅ Found {len(papers)} papers on OpenAlex")
      
       return papers
  
   def scrape_papers_with_code(self): """
       Scrape Papers with Code
       """
      
       print("🔍 Scraping Papers with Code")
      
       url = "https://paperswithcode.com/api/v1/papers/"
       params = {
                                       "page": 1
                                   }
      
       papers = []
      
       while True:
           response = self.session.get(url, params=params)
           data = response.json()
          
           if not data.get("results"):
               break
          
           papers.extend(data[
                                       "results"
                                   ])
          
           if not data.get("next"):
               break
          
           params[
                                       "page"
                                   ] += 1
           time.sleep(1)
      
       self.results[
                                       "papers_with_code"
                                   ] = papers
       print(f"✅ Found {len(papers)} papers on Papers with Code")
      
       return papers
  
   def download_pdf(self, url: str, filename: str): """
       Download PDF from URL
       """
      
       try:
           response = self.session.get(url, stream=True)
          
           with open(filename, 'wb') as f:
               for chunk in response.iter_content(chunk_size=8192):
                   f.write(chunk)
          
           return True
       except Exception as e:
           print(f"Error downloading {url}: {e}")
           return False
  
   def bulk_download_pdfs(self, output_dir: str = "./papers"): """
       Download all PDFs
       """
      
       os.makedirs(output_dir, exist_ok=True)
      
       print("\n📥 Downloading PDFs...")
      
       # Download arXiv PDFs
       for paper in self.results[
                                       "arxiv"
                                   ]:
           if paper.get("pdf_url"):
               arxiv_id = paper[
                                       "id"
                                   ].split("/")[
                                       -1
                                   ]
               filename = f"{output_dir}/arxiv_{arxiv_id}.pdf"
               self.download_pdf(paper[
                                       "pdf_url"
                                   ], filename)
      
       # Download Semantic Scholar open access PDFs
       for paper in self.results[
                                       "semantic_scholar"
                                   ]:
           if paper.get("openAccessPdf"):
               paper_id = paper.get("paperId",
                                   "unknown")
               filename = f"{output_dir}/ss_{paper_id}.pdf"
               self.download_pdf(paper[
                                       "openAccessPdf"
                                   ][
                                       "url"
                                   ], filename)
      
       print("✅ PDF download complete")
  
   def export_results(self, filename: str = "academic_papers.json"): """
       Export all results to JSON
       """
      
       with open(filename, 'w') as f:
           json.dump(self.results, f, indent=2)
      
       print(f"✅ Results exported to {filename}")
  
   def scrape_all(self, query: str): """
       Scrape all databases
       """
      
       print("\n" + "="*80)
       print(f"🚀 SCRAPING ALL ACADEMIC DATABASES FOR: {query}")
       print("="*80)
      
       # Parallel scraping
       with ThreadPoolExecutor(max_workers=4) as executor:
           futures = [
               executor.submit(self.scrape_arxiv, query,
                                       10000),
               executor.submit(self.scrape_semantic_scholar, query,
                                       5000),
               executor.submit(self.scrape_openalex, query),
               executor.submit(self.scrape_papers_with_code)
                                   ]
          
           for future in futures:
               future.result()
      
       print("\n" + "="*80)
       print("✅ SCRAPING COMPLETE")
       print("="*80)
      
       # Print statistics
       total = sum(len(papers) for papers in self.results.values())
       print(f"\n📊 Total papers found: {total}")
       for source, papers in self.results.items():
           print(f"  - {source}: {len(papers)}")
      
       return self.results




# ============================================
# USAGE EXAMPLES
# ============================================


if __name__ == "__main__":
   scraper = UniversalAcademicScraper()
  
   # Example queries
   queries = [
                                       "machine learning",

                                       "transformers",

                                   ]
  
   # Scrape for each query
   for query in queries:
       results = scraper.scrape_all(query)
       scraper.export_results(f"papers_{query.replace(' ', '_')}.json")
      
       # Optional: Download PDFs
       # scraper.bulk_download_pdfs(f"./papers/{query.replace(' ', '_')}")
```
